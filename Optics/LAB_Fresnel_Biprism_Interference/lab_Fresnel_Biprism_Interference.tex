	\section{理论推导}
	\subsection{干涉条纹的形成}
	菲涅耳双棱镜由两个底面相贴、折射角 $\alpha$ 极小的三棱镜组成。当点光源（狭缝）位于双棱镜后的对称轴上时，光线经折射后形成两束具有恒定相位差的相干光，等效于两个虚光源 $S_1$ 和 $S_2$ 的发散。
	
	设狭缝至双棱镜距离为 $a$，双棱镜折射率为 $n$，根据折射定律（小角近似 $\sin \theta \approx \theta$），偏折角 $\delta = (n-1)\alpha$。两虚光源间的距离 $h$ 为：
	\begin{equation}
		h \approx 2a \delta = 2a(n-1)\alpha
	\end{equation}

	\subsection{条纹间距公式}
	设虚光源至观察屏（CCD）的距离为 $L$。在屏上任一点 $P(y)$，两相干光的光程差 $\Delta$ 为：
	\begin{equation}
		\Delta = \sqrt{L^2 + (y + h/2)^2} - \sqrt{L^2 + (y - h/2)^2} \approx \frac{hy}{L}
	\end{equation}
	当 $\Delta = m\lambda$（$m=0, \pm1, \dots$）时出现明纹，其位置 $y_m = m\frac{L\lambda}{h}$。相邻明纹间距 $\Delta x$ 为：
	\begin{equation}
		\Delta x = y_{m+1} - y_m = \frac{L}{h} \lambda \implies \lambda = \frac{h}{L} \Delta x
	\end{equation}

	\subsection{可见度与单色性}
	若光源非严格单色（如 LED），干涉条纹是不同波长条纹的非相干叠加。可见度 $V$ 定义为：
	\begin{equation}
		V = \frac{I_{\max} - I_{\min}}{I_{\max} + I_{\min}}
	\end{equation}
	根据傅里叶光学，对于光谱宽度为 $\Delta \lambda$ 的光源，相干长度 $l_c \approx \lambda^2 / \Delta \lambda$。当光程差超过 $l_c$ 时，条纹可见度将显著下降甚至消失。
	
	\section{实验内容与数据分析}
	
	\subsection{调节光路与规律总结}
	通过调节狭缝、双棱镜与 CCD 的共轴及平行度，观察到干涉条纹。实验总结规律如下：
	\begin{enumerate}
		\item \textbf{狭缝宽度}：狭缝加宽，光强增加但对比度（可见度）下降；过宽时条纹消失。
		\item \textbf{狭缝与双棱镜距离 $a$}：由式(1)知 $a$ 增大则 $h$ 增大，由式(3)知条纹间距 $\Delta x$ 减小。
		\item \textbf{CCD 位置}：增大 $L$（移动相机远离双棱镜），条纹间距 $\Delta x$ 增大，但单位面积光强减弱。
		\item \textbf{棱脊平行度}：若棱脊与狭缝不平行，条纹会发生倾斜且清晰度下降。
	\end{enumerate}

	\subsection{光源单色性影响研究}
	\begin{itemize}
		\item \textbf{纯 LED 源}：光谱带宽 $\Delta \lambda$ 较大（约 \SI{30}{nm}），观察到中央几级条纹清晰，边缘迅速变模糊。
		\item \textbf{加滤色片后}：光谱被窄化，相干长度 $l_c$ 增加，可观察到的条纹数目明显增多，对比度提升。
	\end{itemize}
	
	\section{结论与误差分析}
	\subsection{主要误差来源}
	1. \textbf{原理误差}：$h$ 的测量公式基于薄透镜近似。\\
	2. \textbf{读数误差}：CCD 图像边缘存在噪声，影响像素定位。\\
	3. \textbf{系统误差}：光轴调节不完全共轴导致的条纹畸变。
	
	\subsection{结论}
	本实验利用线性拟合有效提高了波长测量精度。实验证明了光源单色性与干涉场深度（相干长度）的正相关性。
